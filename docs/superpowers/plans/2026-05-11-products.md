# Products Implementation Plan (Plan 3 of 5)

**Spec reference:** `docs/superpowers/specs/2026-05-11-zagros-redesign-design.md` §6.2 (Products index), §6.3 (Product detail), §5.4 (Data display), §5.5 (Filter system), §5.6 (Breadcrumb + furniture).

**Goal:** Make the 14-SKU catalog navigable in under three clicks for an agronomist. Index page = sticky 8-facet sidebar + card grid with two view modes; detail page = full editorial product page (breadcrumb + 2-col header + composition card + compatibility matrix + application rates + use-case strip + related products).

**Dependencies on Plans 1 + 2 (tags `plan-1-foundation`, `plan-2-homepage`):**
- 14 products + tank-mix compatibility data + partners collection
- Editorial primitives (Kicker, SectionNumber, PullQuote, DropCap, PhotoStamp, DisclaimerChip, MonoDataline)
- ReportCard / PartnerCard with locked baselines (used by Related strip)
- Caption strip · Nav · Footer · Layout

**Out of scope (deferred to Plan 4 or 5):**
- /field-reports index + detail
- /partners deep-dive
- About / Contact / Privacy / Terms
- Real product bag photography (Plan 3 ships with SVG bag mockups)
- ⌘K command palette (Plan 6 polish, if scoped)

## Task 7: Build FilterController.astro (client-side filter wiring)

The orchestrator. Reads URL `?` params, applies them to the DOM, listens for facet changes, updates URL, updates count, updates pills, hides/shows cards.

- [ ] **Step 7.1: FilterController.astro** — inline script island.

```astro
---
// No props; just emits the script. Placed once on the index page.
---
<script>
  type FacetState = Record<string, Set<string>>;

  function getState(): FacetState {
    const params = new URLSearchParams(location.search);
    const state: FacetState = {};
    for (const [key, val] of params.entries()) {
      if (!state[key]) state[key] = new Set();
      val.split(',').forEach(v => v && state[key].add(v));
    }
    return state;
  }

  function writeState(state: FacetState) {
    const params = new URLSearchParams();
    for (const [key, values] of Object.entries(state)) {
      if (values.size > 0) params.set(key, Array.from(values).join(','));
    }
    const q = params.toString();
    history.replaceState(null, '', q ? `?${q}` : location.pathname);
  }

  function matchesProduct(card: HTMLElement, state: FacetState, search: string): boolean {
    const text = card.textContent?.toLowerCase() ?? '';
    if (search && !text.includes(search.toLowerCase())) return false;

    for (const [facet, values] of Object.entries(state)) {
      if (facet === 'q') continue;
      if (values.size === 0) continue;
      // Card-level data attributes are precomputed by ProductCard (e.g.,
      // data-facet-application="foliar fertigation"). Test for ANY match.
      const attr = card.dataset[`facet${facet.charAt(0).toUpperCase()}${facet.slice(1)}`] ?? '';
      const cardValues = new Set(attr.split(' ').filter(Boolean));
      let anyMatch = false;
      for (const v of values) if (cardValues.has(v)) { anyMatch = true; break; }
      if (!anyMatch) return false;
    }
    return true;
  }

  function renderPills(state: FacetState) {
    const container = document.querySelector<HTMLElement>('[data-products-pills-container]');
    const wrapper = document.querySelector<HTMLElement>('[data-products-pills]');
    if (!container || !wrapper) return;
    container.innerHTML = '';
    let count = 0;
    for (const [facet, values] of Object.entries(state)) {
      if (facet === 'q') continue;
      for (const v of values) {
        const pill = document.createElement('button');
        pill.type = 'button';
        pill.className = 'inline-flex items-center gap-2 px-3 py-1 border border-ink text-ink bg-paper';
        pill.dataset.removeFacet = facet;
        pill.dataset.removeValue = v;
        pill.innerHTML = `${facet} · ${v} <span aria-hidden="true">×</span>`;
        container.appendChild(pill);
        count++;
      }
    }
    wrapper.classList.toggle('hidden', count === 0);
  }

  function apply() {
    const state = getState();
    const searchInput = document.querySelector<HTMLInputElement>('[data-products-search]');
    const search = searchInput?.value ?? '';
    let visible = 0;
    const cards = document.querySelectorAll<HTMLElement>('[data-product-slug]');
    cards.forEach((card) => {
      const ok = matchesProduct(card, state, search);
      card.toggleAttribute('hidden', !ok);
      if (ok) visible++;
    });
    const countEl = document.querySelector('[data-products-count]');
    if (countEl) countEl.textContent = String(visible);
    renderPills(state);
  }

  function syncCheckboxes() {
    const state = getState();
    document.querySelectorAll<HTMLInputElement>('input[data-facet]').forEach((input) => {
      const facet = input.dataset.facet!;
      const values = state[facet];
      input.checked = !!(values && values.has(input.value));
    });
  }

  document.addEventListener('change', (e) => {
```
